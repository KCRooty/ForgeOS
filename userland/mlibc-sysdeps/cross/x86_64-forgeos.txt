# meson cross-file para ForgeOS (x86_64 bare-metal / ELF)
# Lugar: mlibc/cross/x86_64-forgeos.txt
# Uso:   meson setup build --cross-file cross/x86_64-forgeos.txt

[binaries]
c       = 'clang'
cpp     = 'clang++'
ar      = 'llvm-ar'
nm      = 'nm'
strip   = 'llvm-strip'
objcopy = 'llvm-objcopy'
ld      = 'ld.lld'
pkgconfig = 'pkg-config'

[built-in options]
c_args   = ['-target', 'x86_64-elf',
            '-ffreestanding', '-fno-stack-protector',
            '-mno-red-zone', '-fno-pic',
            '-nostdinc', '-nostdlib',
            '-D_GNU_SOURCE',
            '-I/usr/lib/llvm-19/lib/clang/19/include']
cpp_args = ['-target', 'x86_64-elf',
            '-ffreestanding', '-fno-stack-protector',
            '-mno-red-zone', '-fno-pic',
            '-nostdinc', '-nostdlib',
            '-fno-rtti', '-fno-exceptions',
            '-D_GNU_SOURCE',
            '-I/usr/lib/llvm-19/lib/clang/19/include']
c_link_args   = ['-target', 'x86_64-elf', '-nostdlib', '-fuse-ld=lld', '-rtlib=compiler-rt']
cpp_link_args = ['-target', 'x86_64-elf', '-nostdlib', '-fuse-ld=lld', '-rtlib=compiler-rt']

[host_machine]
system     = 'forgeos'
cpu_family = 'x86_64'
cpu        = 'x86_64'
endian     = 'little'
